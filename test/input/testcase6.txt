program TestAllNodesFixed;

{ --- Uji Node Deklarasi Konstanta & Tipe Data --- }
const
  CInt = 100;
  CReal = 3; 
  CSigned = -50;
  Merah = 1;
  Putih = 2;
  Biru = 3;

type
  TRange = 1..100;
  TEnum = integer;
  TArray = array[1..100] of integer; 
  TRec = record
           FieldX: integer;
           FieldY: integer;
         end;

{ --- Uji Node Deklarasi Variabel & Subprogram --- }
var
  GlobalInt, GlobalCount: integer;
  GlobalReal: integer; 
  GlobalBool: integer; 
  MyArr: TArray;
  MyRec: TRec;
  MyEnum: TEnum;

procedure Hitung(X: integer; Y: integer);
var
  LocalVar: integer;
begin
  LocalVar := X;
end;

function GetStatus(A: integer): integer;
begin
  if A > 0 then
    GetStatus := 1
  else
    GetStatus := 0;
end;

{ --- Uji Node Statement & Ekspresi (Main Program) --- }
begin
  
  { 1. Assignment & Variable Access }
  GlobalInt := 10;
  GlobalReal := 20;
  
  { 2. Component Variable (Array & Record Access) }
  MyArr[1] := 50;           
  MyRec.FieldX := 100;      
  MyArr[GlobalInt] := 5;   

  { 3. Procedure & Function Call }
  Hitung(GlobalInt, GlobalReal);
  GlobalBool := GetStatus(5);

  { 4. If Statement (Relational Operators) }
  if GlobalInt == 10 then
    GlobalInt := 0
  else if GlobalInt <> 0 then
    GlobalInt := 1;

  if GlobalInt >= 5 then
    GlobalReal := 1;
    
  if GlobalReal < 5 then
    GlobalReal := 5;

  { 5. Case Statement }
  MyEnum := Merah;
  case MyEnum of
    Merah: GlobalInt := 1;
    Putih, Biru: GlobalInt := 2;
  end;

  { 6. Loop Statements }
  while GlobalInt < 10 do
  begin
    GlobalInt := GlobalInt + 1;
  end;

  repeat
    GlobalInt := GlobalInt - 1;
  until GlobalInt == 0;

  for GlobalCount := 1 to 10 do
  begin
    GlobalInt := GlobalCount;
  end;

  { Bypass for-downto menggunakan while-do }
  GlobalCount := 10;
  while GlobalCount > 0 do
  begin
    MyArr[GlobalCount] := 0;
    GlobalCount := GlobalCount - 1;
  end;

  { 7. Complex Expressions (Operator Precedence) }
  GlobalInt := 10 * 2;
  GlobalReal := 10 div 2;
  GlobalInt := 10 div 3;
  GlobalInt := 10 mod 3;
  
  GlobalInt := GlobalInt + 1;
  GlobalInt := GlobalInt - 1;
  
  { Bypass ekspresi OR }
  if GlobalInt > 0 then
    GlobalInt := 0
  else if GlobalReal > 0 then
    GlobalInt := 0;

  { Bypass ekspresi NOT }
  if GlobalInt == 0 then
    GlobalBool := 0
  else
    GlobalBool := 1;

end.
